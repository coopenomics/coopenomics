37//
38// Аналог `Marketplace::get_user_wallet_balance` — вынесен отдельно, чтобы не

41struct UserWalletAvailable {
42 eosio::asset available = eosio::asset(0, _root_govern_symbol);
43 eosio::asset blocked = eosio::asset(0, _root_govern_symbol);
44 bool exists = false;
45};

47inline UserWalletAvailable get_user_wallet_balance(eosio::name coopname,
48 eosio::name wallet_id,
49 eosio::name username) {
50 userwallets_index user_wallets(_ledger2, coopname.value);
51 auto idx = user_wallets.get_index<"byuserwallet"_n>();
52 auto it = idx.find(combine_ids(wallet_id.value, username.value));
53 if (it == idx.end()) {
54 return UserWalletAvailable{};
55 }
56 return UserWalletAvailable{ it->available, it->blocked, true };
57}

80inline std::optional<edu_contract> get_contract(eosio::name coopname, eosio::name username) {
81 edu_contracts_index contracts(_edubridge, coopname.value);
82 auto idx = contracts.get_index<"byusername"_n>();

89inline edu_contract get_active_contract_or_fail(eosio::name coopname, eosio::name username) {
90 auto contract = get_contract(coopname, username);
91 eosio::check(contract.has_value(), "EDUBRIDGE_CONTRACT_NOT_SIGNED: Преподаватель не подписал договор участия в хозяйственной деятельности");
92 eosio::check(contract->status == ContractStatus::ACTIVE,
93 "EDUBRIDGE_CONTRACT_NOT_APPROVED: Договор участия в хозяйственной деятельности ещё не подписан председателем совета");
94 return *contract;
95}

108template <typename Fn>
109inline void update_course(eosio::name coopname, uint64_t course_id, Fn&& change) {
110 edu_courses_index courses(_edubridge, coopname.value);
111 auto it = courses.find(course_id);
112 if (it == courses.end()) {
113 courses.emplace(RamPayer::of(courses, coopname), [&](auto& c) {
114 const eosio::asset zero(0, _root_govern_symbol);
115 c.course_id = course_id;
116 c.collected = zero;
117 c.reserve = zero;
118 c.settled = zero;
119 change(c);
120 });
121 } else {
122 courses.modify(it, RamPayer::of(courses, coopname), [&](auto& c) { change(c); });
123 }
124}

134inline void check_money(const eosio::asset& amount, const char* what) {
135 eosio::check(amount.is_valid() && amount.amount > 0,
136 std::string{"EDUBRIDGE_AMOUNT_NOT_POSITIVE: "} + what + " должна быть больше нуля");
137 eosio::check(amount.symbol == _root_govern_symbol,
138 std::string{"EDUBRIDGE_SYMBOL_INVALID: Некорректный символ валюты: "} + what);
139}
