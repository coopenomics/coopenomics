138inline bool is_known_process(eosio::name process_type) {
139 for (const auto& known : PROCESS_REGISTRY) {
140 if (known == process_type) return true;
141 }
142 return false;
143}
144
146constexpr bool process_registry_is_unique() {
147 constexpr auto count = sizeof(PROCESS_REGISTRY) / sizeof(PROCESS_REGISTRY[0]);
148 for (unsigned i = 0; i < count; ++i) {
149 for (unsigned j = i + 1; j < count; ++j) {
150 if (PROCESS_REGISTRY[i].value == PROCESS_REGISTRY[j].value) return false;
151 }
152 }
153 return true;
154}
