80inline asset ext_or_zero(const eosio::binary_extension<asset>& v, const eosio::symbol& sym) {
81 if (!v.has_value()) {
82 return asset(0, sym);
83 }
84 const asset &a = v.value();
85 if (a.symbol == sym) {
86 return a;
87 }
88 eosio::check(a.amount == 0,
89 "program_expense_* содержит ненулевую сумму с несовместимым символом");
90 return asset(0, sym);
91}

140inline void topup_program_expense_pool(eosio::name coopname, const asset &amount) {
141 global_state_table global_state_inst(_capital, _capital.value);
142 auto itr = global_state_inst.find(coopname.value);
143 eosio::check(itr != global_state_inst.end(), "Контракт не инициализирован");
144 eosio::check(amount.is_valid() && amount.amount > 0, "Сумма пополнения должна быть положительной");
145 eosio::check(amount.symbol == _root_govern_symbol, "Символ суммы пополнения должен совпадать с символом программы");
146 eosio::check(itr->global_available_invest_pool >= amount,
147 "Недостаточно свободных инвестиций программы для пополнения пула расходов");
148
149 global_state_inst.modify(itr, _capital, [&](auto &s) {
150 asset pool = ext_or_zero(s.program_expense_pool, amount.symbol);
151 asset reserved = ext_or_zero(s.program_expense_reserved, amount.symbol);
152 s.global_available_invest_pool -= amount;
153 pool += amount;
154 s.program_expense_pool = pool;
155 s.program_expense_reserved = reserved;
156 });
157}
