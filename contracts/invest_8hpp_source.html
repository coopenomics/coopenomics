32inline int64_t returned_of(const mkt_investor& i, uint64_t acc) {
33 const int64_t u = i.units();
34 if (i.status != Status::ACTIVE || u <= 0) return i.returned_fixed;
35 return i.returned_fixed + static_cast<int64_t>(mul(u, acc - i.acc_join) / SCALE);
36}

39inline uint64_t done_threshold(const mkt_investor& i) {
40 const int64_t u = i.units();
41 if (u <= 0) return OUT_OF_RANGE;
42 const int64_t remaining = u - i.returned_fixed;
43 if (remaining <= 0) return i.acc_join;
44 const uint128_t num = static_cast<uint128_t>(remaining) * SCALE;
45 const uint128_t step = (num + static_cast<uint128_t>(u) - 1) / static_cast<uint128_t>(u);
46 return i.acc_join + static_cast<uint64_t>(step);
47}

57inline void leave_base(mkt_invbranch& b, mkt_investor& i) {
58 const int64_t u = i.units();
59 if (i.status == Status::ACTIVE && u > 0) {
60 // В s_fixed лежит прежний returned_fixed; приращение от acc_join до acc
61 // сидит в (acc × base − s_join) и вычитается вместе с base и s_join.
62 b.s_fixed -= i.returned_fixed;
63 b.base -= u;
64 shift_base_money(b, -i.used_money.amount);
65 b.s_join -= mul(u, i.acc_join);
66 i.returned_fixed = returned_of(i, b.acc);
67 }
68 i.acc_join = b.acc;
69}
70
72inline void enter_base(mkt_invbranch& b, mkt_investor& i) {
73 i.acc_join = b.acc;
74 const int64_t u = i.units();
75 if (i.status == Status::ACTIVE && u > 0) {
76 b.base += u;
77 shift_base_money(b, i.used_money.amount);
78 b.s_join += mul(u, b.acc);
79 b.s_fixed += i.returned_fixed;
80 }
81 i.acc_done = done_threshold(i);
82}

86inline int64_t outstanding(const mkt_invbranch& b) {
87 if (b.base <= 0) return 0;
88 const uint128_t accrued = mul(b.base, b.acc) - b.s_join;
89 const int64_t left = b.base - b.s_fixed - static_cast<int64_t>(accrued / SCALE);
90 return left > 0 ? left : 0;
91}

94inline void accept_repay(mkt_invbranch& b, int64_t amount) {
95 b.acc += static_cast<uint64_t>(static_cast<uint128_t>(amount) * SCALE /
96 static_cast<uint128_t>(b.base));

100struct RepayParts {
101 int64_t money = 0;
102 int64_t property = 0;
103 int64_t total() const { return money + property; }
104};
105
108inline RepayParts split_repay(int64_t amount, int64_t money_units, int64_t units) {
109 RepayParts p;
110 if (amount <= 0 || units <= 0) {
111 p.property = amount > 0 ? amount : 0;
112 return p;
113 }
114 const int64_t m = money_units > units ? units : (money_units > 0 ? money_units : 0);
115 p.money = static_cast<int64_t>(static_cast<uint128_t>(amount) * static_cast<uint128_t>(m) /
116 static_cast<uint128_t>(units));
117 p.property = amount - p.money;
118 return p;
119}

126inline RepayParts complete(mkt_invbranch& b, mkt_investor& i) {
127 leave_base(b, i);
128 const int64_t u = i.units();
129 int64_t excess = i.returned_fixed - u;
130 if (excess < 0) excess = 0;
131 i.returned_fixed -= excess;
132 i.status = Status::DONE;
133 i.acc_done = OUT_OF_RANGE;
134 return split_repay(excess, i.used_money.amount, u);
135}

140struct Claimable {
141 int64_t returned = 0;
142 int64_t unused = 0;
143 int64_t total() const { return returned + unused; }
144};

146inline Claimable claimable_of(const mkt_invbranch& b, const mkt_investor& i) {
147 Claimable c;
148 int64_t r = returned_of(i, b.acc);
149 if (r > i.units()) r = i.units();
150 c.returned = r - i.claimed.amount;
151 if (c.returned < 0) c.returned = 0;
152 const bool release_unused = i.status == Status::DONE || b.base <= 0;
153 if (release_unused && i.unused() > 0) {
154 c.unused = i.unused();
155 // Деньги, выделенные в пул и ещё не израсходованные, держат неиспользованное
156 // у любого инициатора — и у завершившего возврат.
157 const int64_t free = b.unused_total.amount - b.reserved.amount;
158 if (c.unused > free) c.unused = free > 0 ? free : 0;
159 }
160 return c;
161}

249inline void add_investment(eosio::name coopname, eosio::name braname, eosio::name username,
250 eosio::asset money, eosio::asset property) {
251 using namespace MarketInvest;
252 mkt_invbranches_index branches(_marketplace, coopname.value);
253 auto bit = branches.find(braname.value);
254 if (bit == branches.end()) {
255 bit = branches.emplace(RamPayer::of(branches, coopname), [&](auto& b) {
256 b.braname = braname;
257 b.money_total = zero_asset();
258 b.property_total = zero_asset();
259 b.unused_total = zero_asset();
260 b.reserved = zero_asset();
261 b.spent_total = zero_asset();
262 b.repaid_total = zero_asset();
263 b.surplus_total = zero_asset();
264 });
265 }
266 mkt_invbranch b = *bit;
267
268 mkt_investors_index investors(_marketplace, coopname.value);
269 auto by_pair = investors.get_index<"bybruser"_n>();
270 auto it = by_pair.find(combine_ids(braname.value, username.value));
271 const bool fresh = it == by_pair.end();
272
273 mkt_investor row;
274 if (fresh) {
275 row.id = investors.available_primary_key();
276 row.braname = braname;
277 row.username = username;
278 row.status = Status::ACTIVE;
279 row.money = zero_asset();
280 row.property = zero_asset();
281 row.used_money = zero_asset();
282 row.claimed = zero_asset();
283 row.join_seq = ++b.join_seq;
284 row.joined_at = eosio::time_point_sec(eosio::current_time_point());
285 } else {
286 row = *it;
287 leave_base(b, row);
288 }
289
290 // До прибавки: у строки прежней сборки запасное значение — текущий остаток.
291 row.money_in = row.money_in_or_current() + money;
292 row.money += money;
293 row.property += property;
294 b.money_total += money;
295 b.property_total += property;
296 b.unused_total += money;
297 // Завершивший возврат возвращается в базу, только когда у него появилось
298 // невозмещённое — новое имущество. Новые деньги до расхода в базу не входят:
299 // он остаётся завершившим и может забрать их заявлением. Иначе он стоял бы в
300 // базе с нулевым остатком к возмещению, завершить возврат было бы нечем, и
301 // его неиспользованные деньги остались бы закрыты.
302 if (row.status == Status::DONE && row.units() > row.returned_fixed) row.status = Status::ACTIVE;
303 enter_base(b, row);
304
305 branches.modify(bit, RamPayer::of(branches, coopname), [&](auto& x) { x = b; });
306 if (fresh) {
307 investors.emplace(RamPayer::of(investors, coopname), [&](auto& x) { x = row; });
308 } else {
309 by_pair.modify(it, RamPayer::of(investors, coopname), [&](auto& x) { x = row; });
310 }
311}

326inline void route_initiator_share(eosio::name actor, eosio::name coopname, eosio::name braname,
327 eosio::asset amount, const checksum256& order_hash) {
328 using namespace MarketInvest;
329 int64_t to_repay = 0;
330 int64_t left = 0;
331
332 mkt_invbranches_index branches(_marketplace, coopname.value);
333 auto bit = branches.find(braname.value);
334 if (bit != branches.end() && bit->base > 0) {
335 left = outstanding(*bit);
336 to_repay = amount.amount < left ? amount.amount : left;
337 }
338
339 if (to_repay > 0) {
340 Ledger2::apply(actor, coopname, operations::marketplace::INVEST_REPAY,
341 processes::marketplace::DUE, eosio::asset(to_repay, _root_govern_symbol),
342 braname, order_hash, Memo::get_invest_repay_memo());
343
344 mkt_invbranch b = *bit;
345 // Каждому инициатору возмещается пропорционально его единицам, поэтому
346 // доля денег во всём возмещении — денежная часть базы к базе.
347 const RepayParts repay = split_repay(to_repay, b.base_money_or_zero(), b.base);
348 if (repay.money > 0)
349 Ledger2::apply(actor, coopname, operations::marketplace::INVEST_COST_COVER,
350 processes::marketplace::DUE, eosio::asset(repay.money, _root_govern_symbol),
351 braname, order_hash, Memo::get_invest_cost_cover_memo());
352
353 accept_repay(b, to_repay);
354 b.repaid_total.amount += to_repay;
355
356 // Завершившие возврат — по индексу от наименьшего порога; каждый шаг
357 // меняет ключ строки на «вне базы», поэтому выборка начинается заново.
358 mkt_investors_index investors(_marketplace, coopname.value);
359 auto by_done = investors.get_index<"bydone"_n>();
360 const uint128_t from = combine_ids(braname.value, 0);
361 const uint128_t upto = combine_ids(braname.value, b.acc);
362 RepayParts excess;
363 for (auto it = by_done.lower_bound(from); it != by_done.end() && it->by_done() <= upto;
364 it = by_done.lower_bound(from)) {
365 mkt_investor row = *it;
366 const RepayParts e = complete(b, row);
367 excess.money += e.money;
368 excess.property += e.property;
369 by_done.modify(it, RamPayer::of(investors, coopname), [&](auto& x) { x = row; });
370 }
371
372 // Списание возместило весь непогашенный остаток участка: в базе не
373 // остаётся никого. Счётчик растёт с округлением вниз, поэтому инициатору
374 // могла остаться единица актива до порога завершения, и возврат завершался
375 // только следующим списанием. Возвращённое таких инициаторов фиксируется
376 // ровно в использованное — переведённая на кошелёк возмещения сумма это
377 // покрывает.
378 if (to_repay == left) {
379 const uint128_t all = combine_ids(braname.value, OUT_OF_RANGE);
380 for (auto it = by_done.lower_bound(from); it != by_done.end() && it->by_done() < all;
381 it = by_done.lower_bound(from)) {
382 mkt_investor row = *it;
383 const RepayParts e = complete(b, row);
384 row.returned_fixed = row.units();
385 excess.money += e.money;
386 excess.property += e.property;
387 by_done.modify(it, RamPayer::of(investors, coopname), [&](auto& x) { x = row; });
388 }
389 }
390
391 // Денежная доля переплаты не может превысить погашенное этим списанием —
392 // остальное считается долей имущества: кредитового сальдо по 26 не бывает.
393 if (excess.money > repay.money) {
394 excess.property += excess.money - repay.money;
395 excess.money = repay.money;
396 }
397 if (excess.total() > 0) {
398 b.surplus_total.amount += excess.total();
399 Ledger2::apply(actor, coopname, operations::marketplace::INVEST_REPAY_SURPLUS,
400 processes::marketplace::DUE, eosio::asset(excess.total(), _root_govern_symbol),
401 braname, order_hash, Memo::get_invest_surplus_memo());
402 if (excess.money > 0)
403 Ledger2::apply(actor, coopname, operations::marketplace::INVEST_COST_UNCOVER,
404 processes::marketplace::DUE, eosio::asset(excess.money, _root_govern_symbol),
405 braname, order_hash, Memo::get_invest_cost_uncover_memo());
406 }
407 branches.modify(bit, RamPayer::of(branches, coopname), [&](auto& x) { x = b; });
408 }
409
410 const int64_t rest = amount.amount - to_repay;
411 if (rest > 0) {
412 Branch::accrue(actor, coopname, braname, eosio::asset(rest, _root_govern_symbol),
413 processes::marketplace::DUE, order_hash,
414 Memo::get_charge_due_distribute_memo());
415 }
