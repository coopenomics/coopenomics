125 constexpr bool account_ids_unique() {
126 for (size_t i = 0; i < LEDGER2_ACCOUNT_MAP_SIZE; ++i) {
127 for (size_t j = i + 1; j < LEDGER2_ACCOUNT_MAP_SIZE; ++j) {
128 if (LEDGER2_ACCOUNT_MAP[i].id == LEDGER2_ACCOUNT_MAP[j].id) return false;
