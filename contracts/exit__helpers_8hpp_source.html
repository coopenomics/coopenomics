33inline void consolidate_share_to_main(name coopname, name username, name wallet_name, asset amount, checksum256 exit_hash) {
34 if (wallet_name == ledger2_wallets::SHARE_FUND_PAY) return; // уже на главном паевом
35
36 eosio::name op;
37 if (wallet_name == ledger2_wallets::MIN_SHARE_FUND) {
38 op = operations::registrator::MOVE_MINSHARE; // w.reg.minshr → w.wal.share
39 } else if (wallet_name == ledger2_wallets::BLAGOROST_FUND) {
40 op = operations::capital::WITHDRAW_FROM_CAPITAL; // w.cap.blago → w.wal.share
41 } else if (wallet_name == ledger2_wallets::MARKETPLACE_SHARE_FUND) {
42 op = operations::marketplace::RECALL_SHARE; // w.mkt.share → w.wal.share
43 } else {
44 eosio::check(false,
45 std::string{"Нет операции консолидации паевого кошелька "} + wallet_name.to_string() +
46 " на главный при выходе — добавьте маппинг в consolidate_share_to_main");
47 }
48
49 // Нитку называет её инициатор — выход из кооператива, поэтому имя одно на
50 // все консолидируемые кошельки. Иначе у одного exit_hash оказалось бы два
51 // имени (p.wal.wthdrw и p.cap.wthcap), и какое победит, зависело бы от того,
52 // на каком кошельке у пайщика ненулевой остаток. Операция возврата из
53 // «Благороста» при этом остаётся собственной операцией — в чужой нитке она
54 // идёт по тому же правилу, что членский взнос КУ внутри поставки.
55 std::string memo = "Консолидация паевого взноса при выходе, кошелёк=" +
56 wallet_name.to_string() + ", username=" + username.to_string();
57 Ledger2::apply(_registrator, coopname, op, processes::wallet::WITHDRAW,
58 amount, username, exit_hash, memo);
59}
