50inline void consolidate_share_to_main(name coopname, name username, name wallet_name, asset amount, checksum256 exit_hash) {
51 if (wallet_name == ledger2_wallets::SHARE_FUND_PAY) return; // уже на главном паевом
52
53 eosio::name op;
54 if (wallet_name == ledger2_wallets::MIN_SHARE_FUND) {
55 op = operations::registrator::MOVE_MINSHARE; // w.reg.minshr → w.wal.share
56 } else if (wallet_name == ledger2_wallets::BLAGOROST_FUND) {
57 op = operations::capital::WITHDRAW_FROM_CAPITAL; // w.cap.blago → w.wal.share
58 } else {
59 eosio::check(false,
60 std::string{"Нет операции консолидации паевого кошелька "} + wallet_name.to_string() +
61 " на главный при выходе — добавьте маппинг в consolidate_share_to_main");
62 }
63
64 // Нитку называет её инициатор — выход из кооператива, поэтому имя одно на
65 // все консолидируемые кошельки. Иначе у одного exit_hash оказалось бы два
66 // имени (p.wal.wthdrw и p.cap.wthcap), и какое победит, зависело бы от того,
67 // на каком кошельке у пайщика ненулевой остаток. Операция возврата из
68 // «Благороста» при этом остаётся собственной операцией — в чужой нитке она
69 // идёт по тому же правилу, что членский взнос КУ внутри поставки.
70 std::string memo = "Консолидация паевого взноса при выходе, кошелёк=" +
71 wallet_name.to_string() + ", username=" + username.to_string();
72 Ledger2::apply(_registrator, coopname, op, processes::wallet::WITHDRAW,
73 amount, username, exit_hash, memo);
74}
