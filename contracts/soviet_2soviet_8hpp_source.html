329 uint64_t primary_key() const { return id; }

332 bool is_expired() const {
333 return expires_at.sec_since_epoch() != 0 && expires_at <= eosio::time_point_sec(eosio::current_time_point().sec_since_epoch());
334 }
335
337 bool allows_vote(eosio::name decision_type) const {
338 if (std::find(vote_types.begin(), vote_types.end(), decision_type) != vote_types.end()) return true;
339 for (const auto& rule : follow_rules) {
340 if (rule.decision_type == decision_type) return true;
341 }
342 return false;
343 }
344
345 bool allows_authorize(eosio::name decision_type) const {
346 return std::find(authorize_types.begin(), authorize_types.end(), decision_type) != authorize_types.end();
347 }
