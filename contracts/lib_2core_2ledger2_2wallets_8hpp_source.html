156// =====================================================================
157// Compile-time валидация реестра.
158// =====================================================================
159//
160// Правила (ADR-002):
161// 1. Имена уникальны.
162// 2. Имена непустые (sentinel `eosio::name{}` запрещён).
163// 3. USER_SHARED-имена соответствуют конвенции `w.<3-char-contract>.<verb>`
164// (символ 0 = 'w', символы 1 и 5 = '.').
165//
166// Полнота поля `kind` обеспечена структурно: в `Ledger2WalletMeta` поле без
167// значения по умолчанию — попытка добавить элемент реестра без явного
168// `WalletKind` ломает сборку.
169namespace ledger2_wallets_detail {
170 // base32 кодирование eosio::name (`.12345abcdefghijklmnopqrstuvwxyz`):
171 // '.' = 0
172 // '1'..'5' = 1..5
173 // 'a'..'z' = 6..31
174 // 64-битное value делится на 12 5-битных позиций (старшие биты — первый символ);
175 // 13-й символ (4 младших бита) для финального реестра не используется (имена ≤ 13 символов).
176 static constexpr uint8_t CHAR_DOT = 0; // '.'
177 static constexpr uint8_t CHAR_W = 28; // 'w' = 6 + ('w' - 'a') = 6 + 22
178
179 constexpr uint8_t decode_char_at(uint64_t value, size_t pos /* 0..11 */) {
180 // pos = 0 — старшие 5 бит (биты 59..63)
181 return static_cast<uint8_t>((value >> (59 - 5 * pos)) & 0x1F);
182 }

200 // ADR-002: USER_SHARED-имена обязаны соответствовать конвенции `w.<contract>.<verb>`.
201 // Конкретно: символ 0 = 'w', символ 1 = '.', символ 5 = '.' (после 3-символьного contract).
202 constexpr bool user_shared_naming_convention() {
203 for (size_t i = 0; i < LEDGER2_WALLET_REGISTRY_SIZE; ++i) {
204 const auto& e = LEDGER2_WALLET_REGISTRY[i];
205 if (e.kind != WalletKind::USER_SHARED) continue;
206 const uint64_t v = e.name.value;
207 if (decode_char_at(v, 0) != CHAR_W) return false; // 'w'
208 if (decode_char_at(v, 1) != CHAR_DOT) return false; // '.'
209 if (decode_char_at(v, 5) != CHAR_DOT) return false; // '.' после 3-char contract
210 }
211 return true;
212 }
213}

321inline uint64_t ledger2_required_program_id(eosio::name wallet_name) {
322 for (const auto& m : LEDGER2_USER_SHARED_PROGRAM_MAPPING) {
323 if (m.wallet_name == wallet_name) return m.required_program_id;
324 }
325 // Кошелёк не в маппинге.
326 for (size_t i = 0; i < LEDGER2_WALLET_REGISTRY_SIZE; ++i) {
327 if (LEDGER2_WALLET_REGISTRY[i].name == wallet_name &&
328 LEDGER2_WALLET_REGISTRY[i].kind == WalletKind::USER_SHARED) {
329 eosio::check(false,
330 std::string{"ledger2_required_program_id: USER_SHARED-кошелёк "} +
331 wallet_name.to_string() +
332 " не имеет записи в LEDGER2_USER_SHARED_PROGRAM_MAPPING");
333 }
334 }
335 return 0; // не USER_SHARED → без проверки
336}
