77void apply_weight_total_delta(eosio::name coopname, eosio::name braname,
78 eosio::name contract, int64_t delta) {
79 if (delta == 0) return;
80
81 branch_weight_totals_index totals(_branch, coopname.value);
82 auto idx = totals.get_index<"bycontrbra"_n>();
83 auto it = idx.find(combine_ids(contract.value, braname.value));
84
85 if (it == idx.end()) {
86 eosio::check(delta > 0, "Системная ошибка: агрегат весов распределения отсутствует");
87 totals.emplace(coopname, [&](auto& t) {
88 t.id = totals.available_primary_key();
89 t.braname = braname;
90 t.contract = contract;
91 t.total_weight = static_cast<uint64_t>(delta);
92 });
93 return;
94 }
95
96 eosio::check(delta > 0 || it->total_weight >= static_cast<uint64_t>(-delta),
97 "Системная ошибка: агрегат весов распределения ушёл бы в минус");
98 const uint64_t new_total = (delta > 0)
99 ? it->total_weight + static_cast<uint64_t>(delta)
100 : it->total_weight - static_cast<uint64_t>(-delta);
101
102 if (new_total == 0) {
103 idx.erase(it);
104 } else {
105 idx.modify(it, coopname, [&](auto& t) { t.total_weight = new_total; });
106 }
107}
