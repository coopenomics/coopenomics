62 return false;

66 void add_due(uint64_t assignment_id, const eosio::asset& amount) {
67 due += amount;
68 for (auto& d : dues) {
69 if (d.assignment_id == assignment_id) { d.amount += amount; return; }
70 }
71 dues.push_back(edu_due{assignment_id, amount});
72 }
73
75 void drop_dues() {

77 dues.clear();
78 }
