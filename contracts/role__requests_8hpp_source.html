89inline std::optional<role_request> get_role_request(eosio::name coopname, const checksum256 &request_hash) {

183inline void approve(
184 eosio::name coopname,
185 uint64_t request_id,
186 const eosio::asset &approved_rate,
187 uint64_t approved_hours
188) {
189 role_requests_index t(_capital, coopname.value);
190 auto itr = t.find(request_id);
191 eosio::check(itr != t.end(), "Заявка не найдена");
192 eosio::check(itr->status == Status::PENDING, "Заявка уже обработана");
193 t.modify(itr, coopname, [&](auto &r) {
194 r.status = Status::APPROVED;
195 r.approved_rate = approved_rate;
196 r.approved_hours = approved_hours;
197 });
198}

200inline void decline(eosio::name coopname, uint64_t request_id, const std::string &reason) {
201 role_requests_index t(_capital, coopname.value);
202 auto itr = t.find(request_id);
203 eosio::check(itr != t.end(), "Заявка не найдена");
204 eosio::check(itr->status == Status::PENDING, "Заявка уже обработана");
205 t.modify(itr, coopname, [&](auto &r) {
206 r.status = Status::DECLINED;
207 r.decline_reason = reason;
208 });
