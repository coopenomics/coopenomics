107inline uint64_t ensure_counter(eosio::name coopname, eosio::name braname, uint32_t period) {
108 const auto found = find_counter_id(coopname, braname, period);
109 if (found.has_value()) return *found;
110 kucounters_index counters(_marketplace, coopname.value);
111 const uint64_t id = counters.available_primary_key();
112 counters.emplace(RamPayer::of(counters), [&](auto& c) {
113 c.id = id;
114 c.braname = braname;
115 c.period = period;
116 c.turnover = eosio::asset(0, _root_govern_symbol);
117 c.secured = eosio::asset(0, _root_govern_symbol);
118 c.retained = eosio::asset(0, _root_govern_symbol);
119 c.orders = 0;
120 c.members = 0;
121 });
122 return id;
123}

210inline void note_return(eosio::name coopname, const order& o, const eosio::asset& cost, const eosio::asset& fee) {
211 if (fee.amount <= 0) return;
212 const uint32_t period = maturity_period(coopname, o.warranty_until);
213 const auto id = find_member_id(coopname, o.orderer, o.delivery_braname, period);
214 eosio::check(id.has_value(),
215 "MARKETPLACE_DUE_CHARGED: Обеспечительная часть по заказу уже списана по итогу периода — возврат с кошелька к списанию невозможен");
216 kudues_index dues(_marketplace, coopname.value);
217 const kudue row = dues.get(*id);
218 eosio::check(row.due >= fee,
219 std::string{"MARKETPLACE_DUE_EXCEEDS: К возврату заявлено больше, чем учтено к списанию: заявлено "} +
220 fee.to_string() + ", учтено " + row.due.to_string());
221 dues.modify(dues.get(*id), RamPayer::of(dues), [&](auto& d) {
222 take(d.turnover, cost);
223 d.due -= fee;
224 });
225 modify_counter_if_open(coopname, o.delivery_braname, period, [&](auto& c) {
226 take(c.turnover, cost);
227 take(c.secured, fee);
228 });
229}

237inline void release_member(eosio::name coopname, eosio::name username) {
238 kudues_index dues(_marketplace, coopname.value);
239 auto by_user = dues.get_index<"byuser"_n>();
240 for (auto it = by_user.lower_bound(username.value);
241 it != by_user.end() && it->username == username;
242 it = by_user.lower_bound(username.value)) {
243 const kudue row = *it;
244 modify_counter_if_open(coopname, row.braname, row.period, [&](auto& c) {
245 take(c.turnover, row.turnover);
246 take(c.secured, row.due);
247 take(c.retained, row.retained);
248 c.orders = c.orders > row.orders ? c.orders - row.orders : 0;
249 c.members = c.members > 0 ? c.members - 1 : 0;
250 });
251 by_user.erase(it);
252 }
253}
