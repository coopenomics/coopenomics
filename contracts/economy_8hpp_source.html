43inline eosio::time_point_sec add_months(eosio::time_point_sec from, uint32_t months, uint32_t month_secs = 0) {
44 // Стенд разработки со сжатым временем: месяц — заданное число секунд.
45 const uint32_t len = month_secs > 0 ? month_secs : SECONDS_IN_MONTH;
46 return eosio::time_point_sec(from.sec_since_epoch() + months * len);

56template <typename Terms>
57inline eosio::time_point_sec program_end(const Terms& terms, eosio::time_point_sec start) {
58 const uint64_t len = terms.month_len() > 0 ? terms.month_len() : SECONDS_IN_MONTH;
59 const uint64_t span = static_cast<uint64_t>(terms.minutes_total) * len / terms.minutes_per_month;
60 return eosio::time_point_sec(start.sec_since_epoch() + static_cast<uint32_t>(span));
61}

147inline FeeQuote quote_fee(const edu_terms& terms, const edu_subscription& sub, eosio::name period, eosio::time_point_sec now) {
148 eosio::check(terms.minutes_per_month > 0 && !terms.slots.empty(),
149 "EDUBRIDGE_COURSE_TERMS_NOT_SET: Условия курса не заданы");
150 const bool paid = sub.has_plan() && sub.plan.value().minutes_paid > 0;
151 const eosio::time_point_sec from = paid && sub.paid_until > now ? sub.paid_until : now;
152
153 FeeQuote q;
154 uint32_t discount_bp = 0;
155 // Сколько минут занятий оплачивает помесячный взнос; меньше месячного числа — взнос по остатку.
156 uint32_t month_minutes = terms.minutes_per_month;
157 if (period == SubscriptionPeriod::MONTH) {
158 const eosio::time_point_sec start = terms.is_started() ? terms.starts_at : from;
159 const eosio::time_point_sec base = from > start ? from : start;
160 q.months = 1;
161 q.paid_from = base;
162 q.paid_until = add_months(base, 1, terms.month_len());
163 const uint32_t total = terms.months_total();
164 if (terms.is_started() && total > 0) {
165 const eosio::time_point_sec end = program_end(terms, start);
166 eosio::check(base < end, "EDUBRIDGE_COURSE_FULLY_PAID: Курс оплачен до конца программы");
167 if (q.paid_until > end) {
168 // Последний неполный месяц: оплачиваются минуты занятий, которые в группе ещё впереди.
169 q.paid_until = end;
170 const uint32_t ahead = terms.minutes_total > terms.minutes_opened ? terms.minutes_total - terms.minutes_opened : 0;
171 eosio::check(ahead > 0, "EDUBRIDGE_COURSE_FULLY_PAID: Все занятия программы проведены");
172 if (ahead < month_minutes) month_minutes = ahead;
173 }
174 }
175 } else {
176 eosio::check(period == SubscriptionPeriod::COURSE, "EDUBRIDGE_PERIOD_INVALID: Недопустимый период оплаты: ожидается month либо course");
177 const uint32_t total = terms.months_total();
178 eosio::check(terms.course_payment && total > 0, "EDUBRIDGE_COURSE_PAYMENT_DISABLED: Взнос за весь курс по этому курсу не принимается");
179 const eosio::time_point_sec start = terms.is_started() ? terms.starts_at : from;
180 const eosio::time_point_sec end = program_end(terms, start);
181 const eosio::time_point_sec base = from > start ? from : start;
182 eosio::check(base < end, "EDUBRIDGE_COURSE_FULLY_PAID: Курс оплачен до конца программы");
183 uint32_t months = 1;
184 while (months < total && add_months(base, months, terms.month_len()) < end) months += 1;
185 q.months = months;
186 q.paid_from = base;
187 q.paid_until = end;
188 discount_bp = terms.discount_bp;
189 }
190
191 // Минут не больше, чем осталось в программе курса.
192 uint32_t minutes = period == SubscriptionPeriod::MONTH ? month_minutes : terms.minutes_per_month * q.months;
193 if (terms.minutes_total > 0) {
194 const uint32_t already = sub.has_plan() ? sub.plan.value().minutes_paid : 0;
195 const uint32_t left = terms.minutes_total > already ? terms.minutes_total - already : 0;
196 eosio::check(left > 0, "EDUBRIDGE_COURSE_FULLY_PAID: Курс оплачен до конца программы");
197 if (minutes > left) minutes = left;
198 }
199 q.minutes = minutes;
200
201 // Взнос идёт за оплаченные минуты: полные месяцы — месячным взносом, неполный
202 // месяц и остаток программы — долей месячного взноса по минутам.
203 eosio::asset base_amount = terms.fee_month() * static_cast<int64_t>(q.months);
204 if (minutes < terms.minutes_per_month * q.months) {
205 base_amount = eosio::asset(
206 static_cast<int64_t>(static_cast<__int128>(terms.fee_month().amount) * minutes / terms.minutes_per_month),
207 base_amount.symbol);
208 }
209 const eosio::asset discount(base_amount.amount * discount_bp / BP_IN_WHOLE, base_amount.symbol);
210 q.amount = base_amount - discount;
211 q.teach = terms.teach_for_minutes(minutes);
212 eosio::check(q.teach <= q.amount, "EDUBRIDGE_DISCOUNT_ABOVE_TARGET_FEE: Скидка курса больше целевого членского взноса");
213 return q;
214}

227inline eosio::asset refusal_refund(const edu_subscription& sub) {
228 const auto& plan = sub.plan.value();
229 const eosio::asset charged = sub.charged_or_zero();
230 if (plan.minutes_paid == 0 || plan.minutes_done >= plan.minutes_paid) return eosio::asset(0, charged.symbol);
231 const int64_t residual = static_cast<int64_t>(
232 static_cast<__int128>(charged.amount) * (plan.minutes_paid - plan.minutes_done) / plan.minutes_paid);
233 return eosio::asset(residual / 2, charged.symbol);
234}

245inline eosio::asset required_lock(const edu_subscription& sub) {
246 const eosio::asset refund = refusal_refund(sub);
247 const eosio::asset reserve = sub.plan.value().reserve;
248 return refund > reserve ? refund - reserve : eosio::asset(0, refund.symbol);
249}

267inline void rebalance_lock(eosio::name coopname, edu_subscription& s) {
268 const eosio::asset locked = s.locked_or_zero();
269 const eosio::asset required = required_lock(s);
270 if (locked > required) {
271 Ledger2::apply(_edubridge, coopname,
272 operations::edubridge::UNLOCK_FEE,
273 processes::edubridge::ACCESS,
274 locked - required, coopname, s.sub_hash,
275 Memo::get_unlock_fee_memo());
276 } else if (locked < required) {
277 Ledger2::apply(_edubridge, coopname,
278 operations::edubridge::LOCK_FEE,
279 processes::edubridge::ACCESS,
280 required - locked, coopname, s.sub_hash,
281 Memo::get_lock_fee_memo());
282 }
283 s.set_amounts(s.charged_or_zero(), s.reserved_or_zero(), required);
284}

294inline void credit_dues(eosio::name coopname, edu_sub_plan& plan) {
295 edu_assignments_index assigns(_edubridge, coopname.value);
296 for (const auto& d : plan.dues) {
297 auto it = assigns.find(d.assignment_id);
298 if (it == assigns.end() || d.amount.amount <= 0) continue;
299 assigns.modify(it, RamPayer::of(assigns, coopname), [&](auto& a) {
300 a.deferred = (a.deferred.symbol == d.amount.symbol ? a.deferred : eosio::asset(0, d.amount.symbol)) + d.amount;
301 });
302 }
303 plan.drop_dues();
304}

326inline bool close_guarantee(eosio::name coopname, const edu_terms& terms, edu_subscription& s, eosio::time_point_sec now) {
327 auto& plan = s.plan.value();
328 // Заявление по гарантийным условиям на рассмотрении совета: взнос заморожен.
329 if (plan.released || plan.claimed || is_guarantee_running(terms, now)) return false;
330
331 const eosio::asset locked = s.locked_or_zero();
332 const eosio::asset required = required_lock(s);
333 // Сначала освобождается удержанное сверх возможного возврата: из него выделяется резерв.
334 if (locked > required) {
335 Ledger2::apply(_edubridge, coopname,
336 operations::edubridge::UNLOCK_FEE,
337 processes::edubridge::ACCESS,
338 locked - required, coopname, s.sub_hash,
339 Memo::get_unlock_fee_memo());
340 }
341
342 const eosio::asset allot = plan.reserve + plan.due;
343 if (allot.amount > 0) {
344 Ledger2::apply(_edubridge, coopname,
345 operations::edubridge::ALLOT_TEACHER_RESERVE,
346 processes::edubridge::ACCESS,
347 allot, coopname, s.sub_hash,
348 Memo::get_allot_reserve_memo());
349 update_course(coopname, s.course_id, [&](auto& c) {
350 c.reserve += allot;
351 check_course_covered(c);
352 });
353 }
354
355 s.set_amounts(s.charged_or_zero(), plan.reserve, locked > required ? required : locked);
356 credit_dues(coopname, plan);
357 plan.released = true;
358 unlock_subscription(coopname, s.course_id);
359 return true;
360}

373inline void settle_closing(eosio::name coopname, const edu_subscription& sub, eosio::asset refund, bool to_share, bool drop_due = false) {
374 const eosio::asset zero(0, _root_govern_symbol);
375 const eosio::asset locked = sub.locked_or_zero();
376 const eosio::asset allotted = sub.reserved_or_zero();
377 // Возврат по гарантийным условиям: участник забирает взнос целиком, в суммы
378 // преподавателей за гарантийный период его занятия не входят.
379 eosio::asset due = sub.has_plan() && !drop_due ? sub.plan.value().due : zero;
380
381 // Возврат участнику обеспечен всегда: его взнос удержан либо лежит в резерве
382 // подписки. Взнос преподавателей за уже проведённые занятия выделяется из
383 // того, что по курсу остаётся после возврата. При полном возврате по
384 // гарантии этого может не хватить — тогда выделяется сколько есть, возврат
385 // не задерживается, а результат преподавателя ждёт средств курса при приёме
386 // (`acceptrid` проверяет резерв курса).
387 if (due.amount > 0) {
388 edu_courses_index courses(_edubridge, coopname.value);
389 auto course = courses.find(sub.course_id);
390 eosio::asset free_after = zero;
391 if (course != courses.end()) {
392 free_after = course->collected - refund - (course->reserve - allotted) - course->settled;
393 }
394 if (free_after.amount < 0) free_after = zero;
395 if (due > free_after) due = free_after;
396 }
397
398 eosio::check(refund.amount >= 0 && refund <= sub.charged_or_zero(), "EDUBRIDGE_REFUND_ABOVE_CHARGED: Возврат больше собранного по подписке");
399
400 if (locked.amount > 0) {
401 Ledger2::apply(_edubridge, coopname,
402 operations::edubridge::UNLOCK_FEE,
403 processes::edubridge::ACCESS,
404 locked, coopname, sub.sub_hash,
405 Memo::get_unlock_fee_memo());
406 }
407 if (allotted.amount > 0) {
408 Ledger2::apply(_edubridge, coopname,
409 operations::edubridge::FREE_TEACHER_RESERVE,
410 processes::edubridge::ACCESS,
411 allotted, coopname, sub.sub_hash,
412 Memo::get_free_reserve_memo());
413 }
414 if (due.amount > 0) {
415 Ledger2::apply(_edubridge, coopname,
416 operations::edubridge::ALLOT_TEACHER_RESERVE,
417 processes::edubridge::ACCESS,
418 due, coopname, sub.sub_hash,
419 Memo::get_allot_reserve_memo());
420 }
421 if (refund.amount > 0) {
422 Ledger2::apply(_edubridge, coopname,
423 operations::edubridge::REFUND_FEE,
424 processes::edubridge::ACCESS,
425 refund, sub.username, sub.sub_hash,
426 Memo::get_refund_memo());
427 if (to_share) {
428 Ledger2::apply(_edubridge, coopname,
429 operations::edubridge::RETURN_TO_SHARE,
430 processes::edubridge::ACCESS,
431 refund, sub.username, sub.sub_hash,
432 Memo::get_return_to_share_memo());
433 }
434 }
435
436 if (allotted.amount > 0 || due.amount > 0 || refund.amount > 0) {
437 update_course(coopname, sub.course_id, [&](auto& c) {
438 eosio::check(allotted <= c.reserve, "EDUBRIDGE_COURSE_RESERVE_INSUFFICIENT: Резерв преподавателям по курсу меньше резерва подписки");
439 c.reserve -= allotted;
440 c.reserve += due;
441 eosio::check(refund <= c.collected, "EDUBRIDGE_REFUND_ABOVE_COLLECTED: Возврат больше собранного по курсу");
442 c.collected -= refund;
443 check_course_covered(c);
444 });
445 }
446
447 if (sub.has_plan()) {
448 const auto& closing = sub.plan.value();
449 if (!closing.released) {
450 // Подписка закрыта до конца гарантийного срока: выделенное за проведённые
451 // занятия переходит в суммы допусков в той доле, какую удалось выделить.
452 if (due.amount > 0) {
453 edu_sub_plan settled = closing;
454 if (due < closing.due) {
455 for (auto& d : settled.dues) {
456 d.amount = eosio::asset(static_cast<int64_t>(static_cast<__int128>(d.amount.amount) * due.amount / closing.due.amount), d.amount.symbol);
457 }
458 }
459 credit_dues(coopname, settled);
460 }
461 unlock_subscription(coopname, sub.course_id);
462 }
463 update_terms(coopname, sub.course_id, [&](auto& t) {
464 if (t.subs_active > 0) t.subs_active -= 1;
465 });
466 }
467}
