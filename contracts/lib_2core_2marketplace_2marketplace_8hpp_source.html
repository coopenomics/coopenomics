60inline void check_packaging(const eosio::asset& quantity, const eosio::asset& package_size) {
61 if (package_size.amount == 0) return; // отпуск по мере
62 eosio::check(package_size.is_valid() && package_size.symbol == quantity.symbol,
63 "Единица упаковки не совпадает с единицей количества");
64 eosio::check(package_size.amount > 0, "Размер упаковки должен быть больше нуля");

66 "Количество должно быть кратно размеру упаковки");

77inline eosio::asset calc_cost(const eosio::asset& quantity, const eosio::asset& unit_price,
78 const eosio::asset& package_size = eosio::asset(0, _unit_piece)) {
79 if (package_size.amount > 0) {
80 const int64_t packages = quantity.amount / package_size.amount; // точно (кратность — check_packaging)
81 const uint128_t total = static_cast<uint128_t>(packages) *
82 static_cast<uint128_t>(unit_price.amount);
83 return eosio::asset(static_cast<int64_t>(total), unit_price.symbol);
84 }
85 int64_t scale = 1;
86 for (uint8_t i = 0; i < quantity.symbol.precision(); ++i) scale *= 10;
87 const uint128_t num = static_cast<uint128_t>(quantity.amount) *
88 static_cast<uint128_t>(unit_price.amount);
89 const int64_t amount = static_cast<int64_t>((num + static_cast<uint128_t>(scale) / 2) / scale);
90 return eosio::asset(amount, unit_price.symbol);
91}

98inline eosio::asset pro_rata(const eosio::asset& total, int64_t part, int64_t whole) {
99 eosio::check(whole > 0, "Некорректная база для расчёта пропорциональной доли");
100 const uint128_t num = static_cast<uint128_t>(total.amount) * static_cast<uint128_t>(part);
101 const int64_t amount =
102 static_cast<int64_t>((num + static_cast<uint128_t>(whole) / 2) / static_cast<uint128_t>(whole));
103 return eosio::asset(amount, total.symbol);
104}

303// ── Cross-contract read: ledger2 wallet/userwallet balances ─────────────
304//
305// Используется в createorder для guard'а Locked Decision L6 (без отрицательного
306// баланса) — проверка достаточности средств заказчика на паевом кошельке
307// перед вызовом o.mkt.lock.
308//
309// ВАЖНО: контракт marketplace не вызывает ledger2::walletop напрямую, а только
310// читает state (RAM-таблицы wallets2 / userwallets через cross-contract scope).
311// Все мутации идут через `Ledger2::apply` (см. lib/core/ledger2/ledger2.hpp).
312
313struct UserWalletAvailable {
314 eosio::asset available = eosio::asset(0, _root_govern_symbol);
315 eosio::asset blocked = eosio::asset(0, _root_govern_symbol);
316 bool exists = false;
317};
318
319inline UserWalletAvailable get_user_wallet_balance(eosio::name coopname,
320 eosio::name wallet_id,
321 eosio::name username) {
322 // userwallets_index — глобальный typedef в lib/domain/table_ledger2_userwallets.hpp.
323 // Для cross-contract read берём scope = coopname.value, code = _ledger2.
324 userwallets_index user_wallets(_ledger2, coopname.value);
325 auto idx = user_wallets.get_index<"byuserwallet"_n>();
326 auto it = idx.find(combine_ids(wallet_id.value, username.value));
327 if (it == idx.end()) {
328 return UserWalletAvailable{};
329 }
330 return UserWalletAvailable{ it->available, it->blocked, true };
331}

382inline void lock_order_body(eosio::name coopname, uint64_t order_id, eosio::name orderer,
383 const checksum256& order_hash, const eosio::asset& body,
384 const std::string& program_memo, const std::string& wallet_memo) {
385 if (body.amount <= 0) return;
386 auto program = get_user_wallet_balance(coopname, ledger2_wallets::MARKETPLACE_SHARE_FUND, orderer);
387 const eosio::asset from_program = program.available >= body ? body : program.available;
388 const eosio::asset from_wallet = body - from_program;
389 if (from_wallet.amount > 0) {
390 auto wallet = get_user_wallet_balance(coopname, ledger2_wallets::SHARE_FUND_PAY, orderer);
391 eosio::check(wallet.available >= from_wallet,
392 std::string{"Недостаточно паевых средств для заказа: требуется с Цифрового кошелька "} +
393 from_wallet.to_string() + ", доступно " + wallet.available.to_string() +
394 " (свободный паевой Стола заказов " + program.available.to_string() + " уже учтён)");
395 }
396 if (from_program.amount > 0) {
397 Ledger2::apply(_marketplace, coopname,
398 operations::marketplace::LOCK_FROM_SHARE,
399 processes::marketplace::SUPPLY,
400 from_program, orderer, order_hash, program_memo);
401 }
402 if (from_wallet.amount > 0) {
403 Ledger2::apply(_marketplace, coopname,
404 operations::marketplace::LOCK_ORDER,
405 processes::marketplace::SUPPLY,
406 from_wallet, orderer, order_hash, wallet_memo);
407 }
408}

477inline void retain_refusal_penalty(eosio::name coopname, const order& o) {
478 // ── Тело заказа 50/50 ──
479 const eosio::asset penalty_body = refusal_penalty_share(o.total_cost);
480 const eosio::asset refund_body = o.total_cost - penalty_body;
481
482 if (refund_body.amount > 0) {
483 Ledger2::apply(_marketplace, coopname,
484 operations::marketplace::UNLOCK_ORDER,
485 processes::marketplace::SUPPLY,
486 refund_body, o.orderer, o.hash,
487 Marketplace::Memo::get_cancel_order_memo(o.id));
488 }
489 if (penalty_body.amount > 0) {
490 // Транзит: удержанная половина тела → пул членских взносов, откуда уйдёт в КУ.
491 Ledger2::apply(_marketplace, coopname,
492 operations::marketplace::REFUSAL_PENALTY,
493 processes::marketplace::SUPPLY,
494 penalty_body, o.orderer, o.hash,
495 Marketplace::Memo::get_refusal_penalty_transit_memo(o.id));
496 }
497
498 // ── Членский взнос 50/50 ──
499 const eosio::asset fee = get_order_membership_fee(o);
500 const eosio::asset penalty_fee = refusal_penalty_share(fee);
501 const eosio::asset refund_fee = fee - penalty_fee;
502 if (refund_fee.amount > 0) {
503 Ledger2::apply(_marketplace, coopname,
504 operations::marketplace::MEMBERSHIP_FEE_REFUND,
505 processes::marketplace::SUPPLY,
506 refund_fee, o.orderer, o.hash,
507 Marketplace::Memo::get_membership_fee_refund_memo(o.id));
508 }

510 // ── Удержанное (тело + взнос) — в общий кошелёк КУ выдачи ──
511 // Обе удержанные половины сейчас в пуле членских взносов: тело — транзитом
512 // выше, взнос — ещё с createorder; единым accrue зачисляются в w.brn.common.
513 const eosio::asset to_common = penalty_body + penalty_fee;
514 if (to_common.amount > 0) {
515 Branch::accrue(_marketplace, coopname, o.delivery_braname,
516 to_common, processes::marketplace::SUPPLY, o.hash,
517 Marketplace::Memo::get_refusal_penalty_distribute_memo(o.id));
518 }
519}

557inline void settle_member_fund_on_exit(eosio::name actor, eosio::name coopname,
558 eosio::name username, const checksum256& exit_hash) {
559 const eosio::asset balance =
560 get_user_wallet_balance(coopname, ledger2_wallets::MARKETPLACE_MEMBER_FUND, username).available;
561 if (balance.amount <= 0) return;
562
563 Ledger2::apply(actor, coopname, operations::marketplace::EXIT_FEE_TO_POOL,
564 processes::wallet::WITHDRAW, balance, username, exit_hash,
565 "Остаток членского кошелька Стола заказов в пул взносов при выходе, username=" +
566 username.to_string());
