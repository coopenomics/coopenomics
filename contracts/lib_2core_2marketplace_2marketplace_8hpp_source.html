78inline eosio::asset calc_cost(const eosio::asset& quantity, const eosio::asset& unit_price,
79 const eosio::asset& package_size = eosio::asset(0, _unit_piece)) {
80 if (package_size.amount > 0) {
81 const int64_t packages = quantity.amount / package_size.amount; // точно (кратность — check_packaging)
82 const uint128_t total = static_cast<uint128_t>(packages) *
83 static_cast<uint128_t>(unit_price.amount);
84 return eosio::asset(static_cast<int64_t>(total), unit_price.symbol);
85 }
86 int64_t scale = 1;
87 for (uint8_t i = 0; i < quantity.symbol.precision(); ++i) scale *= 10;
88 const uint128_t num = static_cast<uint128_t>(quantity.amount) *
89 static_cast<uint128_t>(unit_price.amount);
90 const int64_t amount = static_cast<int64_t>((num + static_cast<uint128_t>(scale) / 2) / scale);
91 return eosio::asset(amount, unit_price.symbol);
92}

99inline eosio::asset pro_rata(const eosio::asset& total, int64_t part, int64_t whole) {
100 eosio::check(whole > 0, "Некорректная база для расчёта пропорциональной доли");
101 const uint128_t num = static_cast<uint128_t>(total.amount) * static_cast<uint128_t>(part);
102 const int64_t amount =
103 static_cast<int64_t>((num + static_cast<uint128_t>(whole) / 2) / static_cast<uint128_t>(whole));
104 return eosio::asset(amount, total.symbol);
105}

304// ── Cross-contract read: ledger2 wallet/userwallet balances ─────────────
305//
306// Используется в createorder для guard'а Locked Decision L6 (без отрицательного
307// баланса) — проверка достаточности свободного паевого «Стола заказов»
308// перед паевым резервом (o.mkt.lockp).
309//
310// ВАЖНО: контракт marketplace не вызывает ledger2::walletop напрямую, а только
311// читает state (RAM-таблицы wallets2 / userwallets через cross-contract scope).
312// Все мутации идут через `Ledger2::apply` (см. lib/core/ledger2/ledger2.hpp).
313
314struct UserWalletAvailable {
315 eosio::asset available = eosio::asset(0, _root_govern_symbol);
316 eosio::asset blocked = eosio::asset(0, _root_govern_symbol);
317 bool exists = false;
318};
319
320inline UserWalletAvailable get_user_wallet_balance(eosio::name coopname,
321 eosio::name wallet_id,
322 eosio::name username) {
323 // userwallets_index — глобальный typedef в lib/domain/table_ledger2_userwallets.hpp.
324 // Для cross-contract read берём scope = coopname.value, code = _ledger2.
325 userwallets_index user_wallets(_ledger2, coopname.value);
326 auto idx = user_wallets.get_index<"byuserwallet"_n>();
327 auto it = idx.find(combine_ids(wallet_id.value, username.value));
328 if (it == idx.end()) {
329 return UserWalletAvailable{};
330 }
331 return UserWalletAvailable{ it->available, it->blocked, true };
332}

435inline void lock_order_body(eosio::name coopname, eosio::name orderer,
436 const checksum256& order_hash, const eosio::asset& body,

445 body.to_string() + ", доступно " + share_left.to_string() +
446 ". Сначала подайте заявление о трансляции паевого взноса в программу.");
447 Ledger2::apply(_marketplace, coopname,
448 operations::marketplace::LOCK_FROM_SHARE,
449 processes::marketplace::SUPPLY,
450 body, orderer, order_hash, memo);
451}

537inline void retain_refusal_penalty(eosio::name coopname, const order& o) {
538 // ── Тело заказа 50/50 ──
539 const eosio::asset penalty_body = refusal_penalty_share(o.total_cost);
540 const eosio::asset refund_body = o.total_cost - penalty_body;
541
542 if (refund_body.amount > 0) {
543 Ledger2::apply(_marketplace, coopname,
544 operations::marketplace::UNLOCK_ORDER,
545 processes::marketplace::SUPPLY,
546 refund_body, o.orderer, o.hash,
547 Marketplace::Memo::get_cancel_order_memo(o.id));
548 }
549 if (penalty_body.amount > 0) {
550 Ledger2::apply(_marketplace, coopname,
551 operations::marketplace::REFUSAL_PENALTY,
552 processes::marketplace::SUPPLY,
553 penalty_body, o.orderer, o.hash,
554 Marketplace::Memo::get_refusal_penalty_memo(o.id));
555 }
556
557 // ── Обеспечительная часть 50/50 ──
558 const eosio::asset fee = get_order_membership_fee(o);
559 const eosio::asset penalty_fee = refusal_penalty_share(fee);
560 refund_secure_part(coopname, o, fee - penalty_fee);
561 move_secure_part_to_due(coopname, o, penalty_fee);
562 // Удержанное списывается участку целиком по окончании текущего периода
563 // (счётчики списания, задача 99D-33).
564 Dues::note_retained(coopname, o, penalty_body + penalty_fee);
565}
