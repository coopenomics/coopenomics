109inline uint32_t utc_month_period(eosio::time_point_sec at) {
110 const int64_t days = static_cast<int64_t>(at.sec_since_epoch()) / 86400;
111 const int64_t z = days + 719468;
112 const int64_t era = (z >= 0 ? z : z - 146096) / 146097;
113 const int64_t doe = z - era * 146097;
114 const int64_t yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365;
115 const int64_t y = yoe + era * 400;
116 const int64_t doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
117 const int64_t mp = (5 * doy + 2) / 153;
118 const int64_t m = mp < 10 ? mp + 3 : mp - 9;
119 const int64_t year = m <= 2 ? y + 1 : y;
120 return static_cast<uint32_t>(year * 12 + m);
121}
