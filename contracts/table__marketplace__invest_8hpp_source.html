143struct [[eosio::table, eosio::contract(MARKETPLACE)]] mkt_investor {
144 uint64_t id;
145 eosio::name braname;
146 eosio::name username;
147 eosio::name status;
148 eosio::asset money;
149 eosio::asset property;
150 eosio::asset used_money;
151 eosio::asset claimed;
152 int64_t returned_fixed = 0;
153 uint64_t acc_join = 0;
154 uint64_t acc_done = 0;
155 uint64_t join_seq = 0;
156 eosio::time_point_sec joined_at;
160 eosio::binary_extension<eosio::asset> money_in;
161
162 uint64_t primary_key() const { return id; }
163 uint64_t by_branch() const { return braname.value; }
164 uint64_t by_user() const { return username.value; }
165 uint128_t by_branch_user() const { return combine_ids(braname.value, username.value); }
166
167 int64_t units() const { return used_money.amount + property.amount; }
169 eosio::asset money_in_or_current() const { return money_in.has_value() ? money_in.value() : money; }
170 int64_t unused() const { return money.amount - used_money.amount; }
171
174 uint128_t by_unused() const {
175 const bool in_queue = unused() > 0;
176 return combine_ids(braname.value, in_queue ? join_seq : MarketInvest::OUT_OF_RANGE);
177 }
179 uint128_t by_done() const {
180 const bool in_base = status == MarketInvest::Status::ACTIVE && units() > 0;
181 return combine_ids(braname.value, in_base ? acc_done : MarketInvest::OUT_OF_RANGE);
182 }
183};
