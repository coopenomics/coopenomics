39struct [[eosio::table, eosio::contract(CAPITAL)]] project {
40 uint64_t id;
41
42 name coopname;
43 checksum256 project_hash;
44 checksum256 parent_hash;
45
46 eosio::name status;
47
48 bool is_opened;
49 bool is_planed;
50 bool is_authorized;
51
52 // Мастер проекта
53 name master;
54
55 std::string title;
56 std::string description;
57 std::string invite;
58 std::string data;
59 std::string meta;
60
61 document2 authorization;
62
63 counts_data counts;
64
65 plan_pool plan;
66 fact_pool fact;
67 crps_data crps;
68
69 // Голосование по методу Водянова
70 voting_data voting;
71
72 // Время создания проекта
73 time_point_sec created_at = current_time_point();
74
75 // ВНИМАНИЕ: поле добавлено в ХВОСТ struct и обёрнуто в binary_extension.
76 // Поля в таблицу EOSIO можно дописывать только в конец struct и только через
77 // binary_extension — иначе ломается десериализация уже записанных проектов.
78 // Контракт стоит на проде, строки projects записаны без этого поля: до
79 // прогона capital::migrate оно приходит как absent и читается через
80 // Capital::Projects::debt_ext_or_zero (absent → 0.0000 RUB).
81 //
82 // Зачем поле: сумма непогашенных ссуд по всем сегментам проекта. Это
83 // производная от segments, а не самостоятельное состояние — она нужна,
84 // чтобы деаллокация (diallocate) могла за O(1) понять, есть ли в проекте
85 // выданные ссуды, и не уронить инвариант provisional_amount >= debt_amount
86 // из update_segment_total_cost. Пересчитывается обходом сегментов в
87 // sync_total_debt: при migrate, при выдаче/отклонении ссуды, при конвертации
88 // сегмента и в самой деаллокации.
89 eosio::binary_extension<eosio::asset> total_debt_amount;
90
91 uint64_t primary_key() const { return id; }
92 uint64_t by_created_at() const { return created_at.sec_since_epoch(); }
93 checksum256 by_hash() const { return project_hash; }
94 checksum256 by_parent_hash() const { return parent_hash; }
95};

113 inline std::optional<project> get_project(eosio::name coopname, const checksum256 &project_hash) {
114 project_index projects(_capital, coopname.value);
115 auto project_hash_index = projects.get_index<"byhash"_n>();
116
117 auto project_itr = project_hash_index.find(project_hash);
118
119 if (project_itr == project_hash_index.end()) {
120 return std::nullopt;
121 }
122
123 return project(*project_itr);
124 }

173 inline void validate_parent_hash(eosio::name coopname, const checksum256 &parent_hash) {
174 checksum256 empty_hash = checksum256();
175
176 if (parent_hash != empty_hash) {
177 auto parent_project = get_project_or_fail(coopname, parent_hash);

272 inline void add_commit(eosio::name coopname, uint64_t project_id, const generation_amounts &delta) {
273 project_index projects(_capital, coopname.value);
274 auto project = projects.find(project_id);
275 eosio::check(project != projects.end(), "Проект не найден");
276 projects.modify(project, _capital, [&](auto &p) {
277 // Увеличиваем счетчик коммитов
278 p.counts.total_commits++;
279
280 // Обновляем время создателей
281 p.fact.creators_hours += delta.creators_hours;
282 // Инкрементальное вычисление среднего для стоимости часа
283 if (delta.hour_cost.amount > 0) {
284 // Расчет скользящего среднего:
285 // новая_средняя = (предыдущая_средняя * (n-1) + новое_значение) / n
286 auto previous_commits = p.counts.total_commits - 1;
287 if (previous_commits == 0) {
288 // Первый коммит - просто устанавливаем значение
289 p.fact.hour_cost = delta.hour_cost;
290 } else {
291 // Последующие коммиты - рассчитываем среднее
292 auto weighted_sum = p.fact.hour_cost.amount * previous_commits;
293 auto new_average = (weighted_sum + delta.hour_cost.amount) / p.counts.total_commits;
294 p.fact.hour_cost = asset(new_average, p.fact.hour_cost.symbol);
295 }
296 }
297
298 // Обновляем остальные пулы
299 p.fact.creators_base_pool += delta.creators_base_pool;
300 p.fact.authors_base_pool += delta.authors_base_pool;
301 p.fact.authors_bonus_pool += delta.authors_bonus_pool;
302 p.fact.creators_bonus_pool += delta.creators_bonus_pool;
303
304 // ВАЖНО: total_generation_pool должен включать все интеллектуальные вклады,
305 // включая уже накопленную базу координаторов и имущество
306 p.fact.total_generation_pool = p.fact.creators_base_pool + p.fact.authors_base_pool +
307 p.fact.creators_bonus_pool + p.fact.authors_bonus_pool +
308 p.fact.coordinators_base_pool + p.fact.property_base_pool;
309
310 // Пересчитываем премии участников от обновленной общей генерации
311 p.fact.contributors_bonus_pool = Capital::Core::Generation::calculate_contributors_bonus_pool(p.fact.total_generation_pool);
312
313 // Общая сумма вкладов пайщиков
314 p.fact.total_contribution = p.fact.total_generation_pool + p.fact.contributors_bonus_pool;
315
316 p.fact.total = p.fact.total_contribution;
317 // Пересчитываем коэффициенты
318 p.fact.return_base_percent = Capital::Core::Generation::calculate_return_base_percent(p.fact.creators_base_pool, p.fact.authors_base_pool, p.fact.coordinators_base_pool, p.fact.invest_pool);
319 p.fact.use_invest_percent = Capital::Core::Generation::calculate_use_invest_percent(p.fact.creators_base_pool, p.fact.authors_base_pool, p.fact.coordinators_base_pool, p.fact.accumulated_expense_pool, p.fact.used_expense_pool, p.fact.total_received_investments);
320
321 // Пересчитываем используемую часть инвестиций и полную стоимость
322 p.fact.total_used_investments = eosio::asset(
323 static_cast<int64_t>(static_cast<double>(p.fact.total_received_investments.amount) * (p.fact.use_invest_percent / 100.0)),
324 _root_govern_symbol
325 );
326 p.fact.total_with_investments = p.fact.total + p.fact.total_used_investments + p.fact.used_expense_pool;
327 });
328 }

385 inline void add_investments(eosio::name coopname, uint64_t project_id, const eosio::asset &amount) {
386 project_index projects(_capital, coopname.value);
387 auto project = projects.find(project_id);
388 eosio::check(project != projects.end(), "Проект не найден");
389
390 projects.modify(project, coopname, [&](auto &p) {
391 // Рассчитываем сколько средств еще нужно для достижения цели по расходам
392 eosio::asset expense_gap = p.plan.target_expense_pool - p.fact.accumulated_expense_pool;
393
394 // Рассчитываем какую часть инвестиций направить в пул расходов
395 eosio::asset to_expense_pool = asset(0, _root_govern_symbol);
396
397 if (expense_gap.amount > 0) {
398 // Рассчитываем процент от инвестиций для пула расходов
399 auto st = Capital::State::get_global_state(coopname);
400 eosio::asset potential_to_expense = amount * st.config.expense_pool_percent / 100;
401
402 // Но не больше, чем нужно для достижения цели
403 to_expense_pool = (potential_to_expense.amount <= expense_gap.amount) ? potential_to_expense : expense_gap;
404 }
405
406 // Остальные средства идут в инвестиционный пул
407 eosio::asset to_invest_pool = amount - to_expense_pool;
408
409 // Обновляем пулы
410 p.fact.invest_pool += to_invest_pool;
411 p.fact.accumulated_expense_pool += to_expense_pool;
412 p.fact.total_received_investments += amount; // Увеличиваем общую сумму полученных инвестиций
413
414 // Пересчитываем коэффициенты возврата
415 p.fact.return_base_percent = Capital::Core::Generation::calculate_return_base_percent(p.fact.creators_base_pool, p.fact.authors_base_pool, p.fact.coordinators_base_pool, p.fact.invest_pool);
416 p.fact.use_invest_percent = Capital::Core::Generation::calculate_use_invest_percent(p.fact.creators_base_pool, p.fact.authors_base_pool, p.fact.coordinators_base_pool, p.fact.accumulated_expense_pool, p.fact.used_expense_pool, p.fact.total_received_investments);
417
418 // Пересчитываем используемую часть инвестиций и полную стоимость
419 p.fact.total_used_investments = eosio::asset(
420 static_cast<int64_t>(static_cast<double>(p.fact.total_received_investments.amount) * (p.fact.use_invest_percent / 100.0)),
421 _root_govern_symbol
422 );
423 p.fact.total_with_investments = p.fact.total + p.fact.total_used_investments + p.fact.used_expense_pool;
424 });
425 }
