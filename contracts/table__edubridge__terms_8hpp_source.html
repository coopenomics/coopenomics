87 static constexpr int64_t WEEKS_PER_MONTH = 4;
88
90 eosio::asset teach_month() const {
91 __int128 week = 0;
92 for (const auto& s : slots) week += s.cost.amount;
93 return eosio::asset(static_cast<int64_t>(week * WEEKS_PER_MONTH), target_fee_month.symbol);
94 }
95
97 eosio::asset teach_for_minutes(uint32_t minutes) const {
98 if (minutes_per_month == 0) return eosio::asset(0, target_fee_month.symbol);
99 return eosio::asset(static_cast<int64_t>(static_cast<__int128>(teach_month().amount) * minutes / minutes_per_month), target_fee_month.symbol);
100 }
101
103 eosio::asset slot_cost(uint32_t slot, uint32_t minutes) const {
104 eosio::check(slot < slots.size(), "EDUBRIDGE_SLOT_NOT_FOUND: Окна расписания с таким номером у группы нет");
105 const auto& s = slots[slot];
106 if (s.minutes == 0 || minutes == s.minutes) return s.cost;
107 return eosio::asset(static_cast<int64_t>(static_cast<__int128>(s.cost.amount) * minutes / s.minutes), s.cost.symbol);
108 }
109
111 eosio::asset fee_month() const {
112 return teach_month() + target_fee_month;
113 }
114
116 uint32_t months_total() const {
117 if (minutes_per_month == 0 || minutes_total == 0) return 0;
118 return (minutes_total + minutes_per_month - 1) / minutes_per_month;
119 }
