71 inline std::optional<program_property> get_program_property(eosio::name coopname, const checksum256 &hash) {
72 program_property_index properties(_capital, coopname.value);
73 auto property_hash_index = properties.get_index<"byhash"_n>();
74
75 auto itr = property_hash_index.find(hash);
76 if (itr == property_hash_index.end()) {
77 return std::nullopt;
78 }
79
80 return program_property(*itr);
81}

151inline void create_program_property_with_approve(
152 eosio::name coopname,
153 eosio::name username,
154 checksum256 property_hash,
155 const eosio::asset &property_amount,
156 const std::string &property_description,
157 const document2 &statement
158) {
159 // Создаем предложение
160 program_property_index properties(_capital, coopname.value);
161 auto property_id = get_global_id_in_scope(_capital, coopname, "pgproperties"_n);
162
163 // Создаем предложение в таблице pgproperties
164 properties.emplace(RamPayer::of(properties, coopname), [&](auto &p) {
165 p.id = property_id;
166 p.status = Capital::ProgramProperties::Status::CREATED;
167 p.coopname = coopname;
168 p.username = username;
169 p.property_hash = property_hash;
170 p.property_amount = property_amount;
171 p.property_description = property_description;
172 p.statement = statement;
173 p.created_at = current_time_point();
174 });
175
176 // Одобрение заявления председателем — под хэшем заявления (ключ одобрения — одобряемый документ).
177 ::Soviet::create_approval(
178 _capital,
179 coopname,
180 username,
181 statement,
182 Names::Capital::CREATE_PROGRAM_PROPERTY,
183 statement.hash,
184 _capital,
185 Names::Capital::APPROVE_PROGRAM_PROPERTY,
186 Names::Capital::DECLINE_PROGRAM_PROPERTY,
187 std::string("")
188 );
189}
