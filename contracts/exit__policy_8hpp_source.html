121 constexpr const ExitWalletRule* find_rule(eosio::name wallet) {
122 for (size_t i = 0; i < EXIT_WALLET_POLICY_SIZE; ++i) {

128 constexpr bool wallets_unique() {

168 constexpr bool transfers_consistent() {
169 for (size_t i = 0; i < EXIT_WALLET_POLICY_SIZE; ++i) {
170 const auto& r = EXIT_WALLET_POLICY[i];
171 if (r.policy != ExitWalletPolicy::RETURN_TO_MAIN) {
172 if (r.transfer_op != eosio::name{}) return false;
173 continue;
174 }
175 const auto* op = find_op(r.transfer_op);
176 if (!op) return false;
177 if (op->wallet_op != WalletOp::TRANSFER) return false;
178 if (op->wallet_from != r.wallet) return false;
179 if (op->wallet_to != ledger2_wallets::SHARE_FUND_PAY) return false;
180 }
181 return true;
182 }

185 constexpr bool single_main() {
186 size_t count = 0;
187 for (size_t i = 0; i < EXIT_WALLET_POLICY_SIZE; ++i) {
188 if (EXIT_WALLET_POLICY[i].policy == ExitWalletPolicy::MAIN) ++count;
189 }
190 return count == 1;
191 }
