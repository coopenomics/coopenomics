193// Удаление BLOCK/UNBLOCK/BURN_BLOCKED (2026-05-24): механика «заблокированного»
194// баланса упразднена. Резерв средств под заявку на возврат паевого теперь
195// выражается переводом на отдельный кошелёк-резерв `w.wal.wpend` (TRANSFER),
196// возврат резерва — обратным TRANSFER, завершение — BURN с резерва. Поле
197// `blocked` в таблицах wallets2/userwallets оставлено deprecated (всегда 0
198// после ledger2::migrate-свёртки) — физическое удаление поля = небезопасная
199// смена layout таблицы на живых коопах, выносится в отдельный cleanup-деплой.
200//
201// Числовые значения ISSUE/TRANSFER/BURN/NONE СОХРАНЕНЫ (не перенумерованы),
202// чтобы исторические op_code в blockchain_actions читались бэкендом без сдвига
203// смысла (2 и 3 — бывшие BLOCK/UNBLOCK — больше не выдаются и невалидны на входе).
204enum class WalletOp : uint8_t {
205 ISSUE = 0,
206 TRANSFER = 1,
207 BURN = 4,
208 NONE = 5,
209};

759// =====================================================================
760// Compile-time валидация реестра.
761// =====================================================================
762//
763// Правила (ADR-003):
764// 1. `code` уникален. `process_type` может повторяться.
765// 2. `(debit_account_id == 0) ⇔ (credit_account_id == 0)` — без частичных проводок.
766// 3. Для записей с проводками (оба ≠ 0): `debit_account_id` ≠ `credit_account_id`,
767// оба существуют в `LEDGER2_ACCOUNT_MAP`.
768// 4. Для TRANSFER: `wallet_from` ≠ `wallet_to`, оба ≠ 0.
769// 5. Для ISSUE: `wallet_from` == 0 и `wallet_to` ≠ 0.
770// 6. Для BURN: `wallet_from` ≠ 0, `wallet_to` == 0.
771// 7. Все id кошельков из записей существуют в `LEDGER2_WALLET_REGISTRY`.
772namespace ledger2_registry_detail {
773 constexpr bool operation_codes_unique() {
774 for (size_t i = 0; i < OPERATION_REGISTRY_SIZE; ++i) {
775 for (size_t j = i + 1; j < OPERATION_REGISTRY_SIZE; ++j) {
776 if (OPERATION_REGISTRY[i].code == OPERATION_REGISTRY[j].code) return false;
777 }
778 }
779 return true;
780 }

782 // Правило 2: без частичных проводок. Либо оба == 0 (без бухпроводки), либо оба ≠ 0.
783 constexpr bool zero_accounts_iff_both() {
784 for (size_t i = 0; i < OPERATION_REGISTRY_SIZE; ++i) {
785 const auto& e = OPERATION_REGISTRY[i];
786 const bool dr_zero = (e.debit_account_id == 0);
787 const bool cr_zero = (e.credit_account_id == 0);
788 if (dr_zero != cr_zero) return false;
789 }
790 return true;
791 }

793 // Правило 3: при наличии проводки — debit ≠ credit.
794 constexpr bool dr_ne_cr_when_posting() {
795 for (size_t i = 0; i < OPERATION_REGISTRY_SIZE; ++i) {
796 const auto& e = OPERATION_REGISTRY[i];
797 if (e.debit_account_id == 0 && e.credit_account_id == 0) continue; // без проводок
798 if (e.debit_account_id == e.credit_account_id) return false;
799 }
800 return true;
801 }

803 // Правило 4: TRANSFER — wallet_from ≠ wallet_to, оба ≠ 0.
804 constexpr bool transfer_wallet_from_ne_to() {
805 for (size_t i = 0; i < OPERATION_REGISTRY_SIZE; ++i) {
806 const auto& e = OPERATION_REGISTRY[i];
807 if (e.wallet_op != WalletOp::TRANSFER) continue;
808 if (e.wallet_from == e.wallet_to) return false;
809 if (e.wallet_from.value == 0 || e.wallet_to.value == 0) return false;
810 }
811 return true;
812 }

814 // Правило 6: BURN — wallet_from required, wallet_to == 0 (ADR-003).
815 constexpr bool burn_pattern_correct() {
816 for (size_t i = 0; i < OPERATION_REGISTRY_SIZE; ++i) {
817 const auto& e = OPERATION_REGISTRY[i];
818 if (e.wallet_op != WalletOp::BURN) continue;
819 if (e.wallet_from.value == 0) return false;
820 if (e.wallet_to.value != 0) return false;
821 }
822 return true;
823 }
824
825 // Правило 8: NONE — оба wallet пустые, обе проводки обязательны (Dr ≠ 0, Cr ≠ 0).
826 // Семантика: только бухпроводка, кошельковое движение отсутствует.
827 constexpr bool none_pattern_correct() {
828 for (size_t i = 0; i < OPERATION_REGISTRY_SIZE; ++i) {
829 const auto& e = OPERATION_REGISTRY[i];
830 if (e.wallet_op != WalletOp::NONE) continue;
831 if (e.wallet_from.value != 0) return false;
832 if (e.wallet_to.value != 0) return false;
833 if (e.debit_account_id == 0) return false;
834 if (e.credit_account_id == 0) return false;
835 }
836 return true;
837 }

840 constexpr bool accounts_exist_in_map() {
841 for (size_t i = 0; i < OPERATION_REGISTRY_SIZE; ++i) {
842 const auto& e = OPERATION_REGISTRY[i];
843 if (e.debit_account_id == 0 && e.credit_account_id == 0) continue; // без проводок

845 if (ledger2_find_account_meta(e.credit_account_id) == nullptr) return false;

855 if (e.wallet_to.value != 0 && !ledger2_is_known_wallet(e.wallet_to)) return false;
