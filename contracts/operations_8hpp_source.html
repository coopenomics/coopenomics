1372// =====================================================================
1373// Compile-time валидация реестра.
1374// =====================================================================
1375//
1376// Правила (ADR-003):
1377// 1. `code` уникален. `process_type` может повторяться.
1378// 2. `(debit_account_id == 0) ⇔ (credit_account_id == 0)` — без частичных проводок.
1379// 3. Для записей с проводками (оба ≠ 0): `debit_account_id` ≠ `credit_account_id`,
1380// оба существуют в `LEDGER2_ACCOUNT_MAP`.
1381// 4. Для TRANSFER: `wallet_from` ≠ `wallet_to`, оба ≠ 0.
1382// 5. Для ISSUE: `wallet_from` == 0 и `wallet_to` ≠ 0.
1383// 6. Для BURN: `wallet_from` ≠ 0, `wallet_to` == 0.
1384// 7. Все id кошельков из записей существуют в `LEDGER2_WALLET_REGISTRY`.
1385namespace ledger2_registry_detail {
1386 constexpr bool operation_codes_unique() {
1387 for (size_t i = 0; i < OPERATION_REGISTRY_SIZE; ++i) {
1388 for (size_t j = i + 1; j < OPERATION_REGISTRY_SIZE; ++j) {
1389 if (OPERATION_REGISTRY[i].code == OPERATION_REGISTRY[j].code) return false;
1390 }
1391 }
1392 return true;
1393 }
1394
1395 // Правило 2: без частичных проводок. Либо оба == 0 (без бухпроводки), либо оба ≠ 0.
1396 constexpr bool zero_accounts_iff_both() {
1397 for (size_t i = 0; i < OPERATION_REGISTRY_SIZE; ++i) {
1398 const auto& e = OPERATION_REGISTRY[i];
1399 const bool dr_zero = (e.debit_account_id == 0);
1400 const bool cr_zero = (e.credit_account_id == 0);
1401 if (dr_zero != cr_zero) return false;
1402 }
1403 return true;
1404 }
1405
1406 // Правило 3: при наличии проводки — debit ≠ credit.
1407 constexpr bool dr_ne_cr_when_posting() {
1408 for (size_t i = 0; i < OPERATION_REGISTRY_SIZE; ++i) {
1409 const auto& e = OPERATION_REGISTRY[i];
1410 if (e.debit_account_id == 0 && e.credit_account_id == 0) continue; // без проводок
1411 if (e.debit_account_id == e.credit_account_id) return false;
1412 }
1413 return true;
1414 }
1415
1416 // Правило 4: TRANSFER — wallet_from ≠ wallet_to, оба ≠ 0.
1417 constexpr bool transfer_wallet_from_ne_to() {
1418 for (size_t i = 0; i < OPERATION_REGISTRY_SIZE; ++i) {
1419 const auto& e = OPERATION_REGISTRY[i];
1420 if (e.wallet_op != WalletOp::TRANSFER) continue;
1421 if (e.wallet_from == e.wallet_to) return false;
1422 if (e.wallet_from.value == 0 || e.wallet_to.value == 0) return false;
1423 }
1424 return true;
1425 }
1426
1427 // Правило 6: BURN — wallet_from required, wallet_to == 0 (ADR-003).
1428 constexpr bool burn_pattern_correct() {
1429 for (size_t i = 0; i < OPERATION_REGISTRY_SIZE; ++i) {
1430 const auto& e = OPERATION_REGISTRY[i];
1431 if (e.wallet_op != WalletOp::BURN) continue;
1432 if (e.wallet_from.value == 0) return false;
1433 if (e.wallet_to.value != 0) return false;
1434 }
1435 return true;
1436 }
1437
1438 // Правило 8: NONE — оба wallet пустые, обе проводки обязательны (Dr ≠ 0, Cr ≠ 0).
1439 // Семантика: только бухпроводка, кошельковое движение отсутствует.
1440 constexpr bool none_pattern_correct() {
1441 for (size_t i = 0; i < OPERATION_REGISTRY_SIZE; ++i) {
1442 const auto& e = OPERATION_REGISTRY[i];
1443 if (e.wallet_op != WalletOp::NONE) continue;
1444 if (e.wallet_from.value != 0) return false;
1445 if (e.wallet_to.value != 0) return false;
1446 if (e.debit_account_id == 0) return false;
1447 if (e.credit_account_id == 0) return false;
1448 }
1449 return true;
1450 }
1451
1452 // Правило 3: оба account_id (если ≠ 0) существуют в LEDGER2_ACCOUNT_MAP.
1453 constexpr bool accounts_exist_in_map() {
1454 for (size_t i = 0; i < OPERATION_REGISTRY_SIZE; ++i) {
1455 const auto& e = OPERATION_REGISTRY[i];
1456 if (e.debit_account_id == 0 && e.credit_account_id == 0) continue; // без проводок
1457 if (ledger2_find_account_meta(e.debit_account_id) == nullptr) return false;
1458 if (ledger2_find_account_meta(e.credit_account_id) == nullptr) return false;
1459 }
1460 return true;
1461 }
1462
1463 // Правило 7: все ссылки на кошельки существуют в реестре.
1464 constexpr bool wallets_exist_in_registry() {
1465 for (size_t i = 0; i < OPERATION_REGISTRY_SIZE; ++i) {
1466 const auto& e = OPERATION_REGISTRY[i];
1467 if (e.wallet_from.value != 0 && !ledger2_is_known_wallet(e.wallet_from)) return false;
1468 if (e.wallet_to.value != 0 && !ledger2_is_known_wallet(e.wallet_to)) return false;
1469 }
1470 return true;
1471 }
